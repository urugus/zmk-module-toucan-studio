if(CONFIG_TOUCAN_STUDIO_POINTER_SPEED OR CONFIG_TOUCAN_STUDIO_SCROLL OR CONFIG_TOUCAN_STUDIO_RIGHT_TRACKPAD)
  zephyr_library()
  zephyr_include_directories(include)
  # A19: an id that differs per build and a short label. firmware/build.sh passes both; other
  # builds (a user's GitHub Actions, M17) get a random id here and a label from the commit (D52).
  set(tsp_gen ${CMAKE_CURRENT_BINARY_DIR}/tsp_generated)
  if(NOT DEFINED TOUCAN_BUILD_ID)
    string(RANDOM LENGTH 16 ALPHABET 0123456789abcdef TOUCAN_BUILD_ID)
  endif()
  if(NOT DEFINED TOUCAN_BUILD_LABEL)
    string(TIMESTAMP tsp_date "%Y-%m-%d" UTC)
    if(DEFINED ENV{GITHUB_SHA})
      string(SUBSTRING "$ENV{GITHUB_SHA}" 0 7 tsp_sha)
      set(TOUCAN_BUILD_LABEL "${tsp_sha} ${tsp_date}")
    else()
      set(TOUCAN_BUILD_LABEL "local ${tsp_date}")
    endif()
  endif()
  if(NOT TOUCAN_BUILD_ID MATCHES "^[0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f]$")
    message(FATAL_ERROR "TOUCAN_BUILD_ID must be 16 lowercase hex digits")
  endif()
  string(REGEX REPLACE "(..)" "0x\\1," tsp_id "${TOUCAN_BUILD_ID}")
  set(tsp_id "{${tsp_id}}")
  set(tsp_label "${TOUCAN_BUILD_LABEL}")
  if(NOT tsp_label MATCHES "^[ -~]*$" OR tsp_label MATCHES "[\"\\\\]")
    message(FATAL_ERROR "TOUCAN_BUILD_LABEL must be printable ASCII without quotes or backslashes")
  endif()
  file(WRITE ${tsp_gen}/tsp_build_info_values.h
    "#define TSP_BUILD_ID ${tsp_id}\n#define TSP_BUILD_LABEL \"${tsp_label}\"\n")
  zephyr_include_directories(${tsp_gen})
  zephyr_library_sources(src/build_info.c)
  zephyr_library_sources_ifdef(CONFIG_TOUCAN_STUDIO_POINTER_SPEED src/pointer_speed.c)
  zephyr_library_sources_ifdef(CONFIG_TOUCAN_STUDIO_SCROLL src/scroll.c)
  zephyr_library_sources_ifdef(CONFIG_TOUCAN_STUDIO_RAW_HID src/raw_hid_protocol.c)
  zephyr_library_sources_ifdef(CONFIG_TOUCAN_STUDIO_RIGHT_RELAY src/right_relay.c)
  zephyr_library_sources_ifdef(CONFIG_TOUCAN_STUDIO_RIGHT_TRACKPAD src/right_trackpad.c src/right_client.c)
endif()
